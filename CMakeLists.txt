# Separate modules remove irrelevant BC3 recovery code from BC1 production.
foreach(codec bc1 bc3)
  string(TOUPPER ${codec} codec_define)
  if(codec STREQUAL "bc1")
    # Select the exact weighted-only BC1 specialization tested on Pixel.
    set(codec_source src/bc_etc2_encode.comp)
    set(codec_options -DPVB_ETC_BC1_ONLY=1)
  else()
    set(codec_source src/bc_etc2_encode.comp)
    set(codec_options -DPVB_ETC_${codec_define}_ONLY=1)
  endif()
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/bc_etc2_encode_${codec}.spv ${CMAKE_CURRENT_BINARY_DIR}/bc_etc2_encode_${codec}_spv.h
    COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 ${codec_options} -S comp -o ${CMAKE_CURRENT_BINARY_DIR}/bc_etc2_encode_${codec}.spv ${CMAKE_CURRENT_SOURCE_DIR}/${codec_source}
    COMMAND ${SPIRV_VALIDATOR} --target-env vulkan1.1 ${CMAKE_CURRENT_BINARY_DIR}/bc_etc2_encode_${codec}.spv
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_spirv.py ${CMAKE_CURRENT_BINARY_DIR}/bc_etc2_encode_${codec}.spv ${CMAKE_CURRENT_BINARY_DIR}/bc_etc2_encode_${codec}_spv.h bc_etc2_encode_${codec}_spv
    DEPENDS ${codec_source} scripts/embed_spirv.py VERBATIM)
endforeach()
