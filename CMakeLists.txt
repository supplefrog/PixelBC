cmake_minimum_required(VERSION 3.22)
project(pixel_vulkan_wrapper VERSION 0.1.2 LANGUAGES CXX)
add_compile_definitions(PVB_VERSION="${PROJECT_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=. -ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=build -fvisibility-inlines-hidden)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
find_program(SPIRV_VALIDATOR spirv-val REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT VULKAN_HEADERS_DIR)
  if(ANDROID)
    set(VULKAN_HEADERS_DIR "${CMAKE_SYSROOT}/usr/include")
  else()
    find_path(VULKAN_HEADERS_DIR vulkan/vulkan.h REQUIRED)
  endif()
endif()
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/bc_decode.spv ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h
  COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 -S comp -o ${CMAKE_CURRENT_BINARY_DIR}/bc_decode.spv ${CMAKE_CURRENT_SOURCE_DIR}/src/bc_decode.comp
  COMMAND ${SPIRV_VALIDATOR} --target-env vulkan1.1 ${CMAKE_CURRENT_BINARY_DIR}/bc_decode.spv
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_spirv.py ${CMAKE_CURRENT_BINARY_DIR}/bc_decode.spv ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h
  DEPENDS src/bc_decode.comp scripts/embed_spirv.py VERBATIM)
add_library(vulkan_pixel_bc SHARED src/wrapper.cpp ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
target_include_directories(vulkan_pixel_bc PRIVATE ${VULKAN_HEADERS_DIR} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(vulkan_pixel_bc PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers -fvisibility=hidden)
target_link_libraries(vulkan_pixel_bc PRIVATE dl)
target_link_options(vulkan_pixel_bc PRIVATE -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map)
set_property(TARGET vulkan_pixel_bc APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/exports.map)
if(ANDROID)
  target_link_libraries(vulkan_pixel_bc PRIVATE log)
  target_link_options(vulkan_pixel_bc PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384 -Wl,-z,nodelete)
else()
  target_link_libraries(vulkan_pixel_bc PRIVATE pthread)
endif()
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/sample.spv ${CMAKE_CURRENT_BINARY_DIR}/sample_spv.h
  COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 -S comp -o ${CMAKE_CURRENT_BINARY_DIR}/sample.spv ${CMAKE_CURRENT_SOURCE_DIR}/tests/sample.comp
  COMMAND ${SPIRV_VALIDATOR} --target-env vulkan1.1 ${CMAKE_CURRENT_BINARY_DIR}/sample.spv
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_spirv.py ${CMAKE_CURRENT_BINARY_DIR}/sample.spv ${CMAKE_CURRENT_BINARY_DIR}/sample_spv.h sample_spv
  DEPENDS tests/sample.comp scripts/embed_spirv.py VERBATIM)
add_executable(pixel_bc_probe tests/probe.cpp ${CMAKE_CURRENT_BINARY_DIR}/sample_spv.h ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
target_include_directories(pixel_bc_probe PRIVATE ${VULKAN_HEADERS_DIR} src tests ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pixel_bc_probe PRIVATE dl)
target_compile_options(pixel_bc_probe PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
if(ANDROID)
  target_link_options(pixel_bc_probe PRIVATE -Wl,-z,max-page-size=16384)
endif()
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample.spv ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample_spv.h
  COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.1 -S comp -o ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample.spv ${CMAKE_CURRENT_SOURCE_DIR}/tests/bc7_cube_sample.comp
  COMMAND ${SPIRV_VALIDATOR} --target-env vulkan1.1 ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample.spv
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_spirv.py ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample.spv ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample_spv.h bc7_cube_sample_spv
  DEPENDS tests/bc7_cube_sample.comp scripts/embed_spirv.py VERBATIM)
add_executable(pixel_bc7_probe tests/bc7_probe.cpp ${CMAKE_CURRENT_BINARY_DIR}/sample_spv.h ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h ${CMAKE_CURRENT_BINARY_DIR}/bc7_cube_sample_spv.h)
target_include_directories(pixel_bc7_probe PRIVATE ${VULKAN_HEADERS_DIR} src tests ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pixel_bc7_probe PRIVATE dl)
target_compile_options(pixel_bc7_probe PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
if(ANDROID)
  target_link_options(pixel_bc7_probe PRIVATE -Wl,-z,max-page-size=16384)
endif()
enable_testing()
if(NOT ANDROID)
  add_executable(transfer_ownership tests/transfer_ownership.cpp ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
  target_include_directories(transfer_ownership PRIVATE ${VULKAN_HEADERS_DIR} src ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(transfer_ownership PRIVATE dl pthread)
  target_compile_options(transfer_ownership PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
  add_test(NAME wrapper_transfer_ownership COMMAND transfer_ownership)
  add_executable(diagnostic_failures tests/diagnostic_failures.cpp ${CMAKE_CURRENT_BINARY_DIR}/sample_spv.h ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
  target_include_directories(diagnostic_failures PRIVATE ${VULKAN_HEADERS_DIR} src tests ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(diagnostic_failures PRIVATE dl)
  target_compile_options(diagnostic_failures PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
  add_test(NAME diagnostic_failures COMMAND diagnostic_failures)
  add_executable(android_backend_lifetime tests/android_backend_lifetime.cpp ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
  target_include_directories(android_backend_lifetime PRIVATE ${VULKAN_HEADERS_DIR} src tests/android_shim ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(android_backend_lifetime PRIVATE dl pthread)
  target_compile_options(android_backend_lifetime PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
  add_test(NAME android_backend_lifetime COMMAND android_backend_lifetime)
  foreach(char_mode signed unsigned)
    add_executable(reference_${char_mode}_char tests/reference_portability.cpp)
    target_compile_options(reference_${char_mode}_char PRIVATE -f${char_mode}-char -Wall -Wextra -Werror)
    add_test(NAME reference_${char_mode}_char COMMAND reference_${char_mode}_char)
  endforeach()
  add_library(probe_lifecycle_backend SHARED tests/lifecycle_backend.cpp)
  target_include_directories(probe_lifecycle_backend PRIVATE ${VULKAN_HEADERS_DIR} src)
  add_executable(probe_lifecycle tests/probe_lifecycle.cpp ${CMAKE_CURRENT_BINARY_DIR}/sample_spv.h ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
  target_include_directories(probe_lifecycle PRIVATE ${VULKAN_HEADERS_DIR} src tests ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(probe_lifecycle PRIVATE dl)
  add_test(NAME probe_lifecycle COMMAND probe_lifecycle $<TARGET_FILE:probe_lifecycle_backend> ${CMAKE_CURRENT_BINARY_DIR}/probe-lifecycle.log)
  add_executable(ownership_failures tests/ownership_failures.cpp ${CMAKE_CURRENT_BINARY_DIR}/bc_decode_spv.h)
  target_include_directories(ownership_failures PRIVATE ${VULKAN_HEADERS_DIR} ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(ownership_failures PRIVATE dl pthread)
  target_compile_options(ownership_failures PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
  add_test(NAME wrapper_ownership_failures COMMAND ownership_failures)
  add_test(NAME wrapper_gpu_probe COMMAND pixel_bc_probe $<TARGET_FILE:vulkan_pixel_bc>)
  add_test(NAME wrapper_bc7_probe COMMAND pixel_bc7_probe $<TARGET_FILE:vulkan_pixel_bc> --validation)
endif()
